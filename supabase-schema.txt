create table if not exists public.equipment_material_app_state (
  id text primary key,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);

alter table public.equipment_material_app_state enable row level security;

create table if not exists public.equipment_material_user_roles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  email text not null,
  role text not null default 'viewer' check (role in ('viewer', 'editor', 'admin')),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table public.equipment_material_user_roles enable row level security;

create table if not exists public.equipment_material_backups (
  id uuid primary key default gen_random_uuid(),
  backup_date text not null unique default to_char(current_date, 'YYYY-MM-DD'),
  created_at timestamptz not null default now(),
  created_by text,
  data jsonb not null
);

alter table public.equipment_material_backups enable row level security;

create or replace function public.equipment_material_current_role()
returns text
language sql
stable
security definer
set search_path = public
as $$
  select coalesce((select role from public.equipment_material_user_roles where user_id = auth.uid()), 'viewer');
$$;

create or replace function public.equipment_material_handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into public.equipment_material_user_roles (user_id, email, role)
  values (
    new.id,
    coalesce(new.email, ''),
    case when lower(coalesce(new.email, '')) = 'jasonb@neearth.com' then 'admin' else 'viewer' end
  )
  on conflict (user_id) do update set email = excluded.email;
  return new;
end;
$$;

drop trigger if exists equipment_material_on_auth_user_created on auth.users;
create trigger equipment_material_on_auth_user_created
after insert or update of email on auth.users
for each row execute function public.equipment_material_handle_new_user();

insert into public.equipment_material_user_roles (user_id, email, role)
select id, coalesce(email, ''), case when lower(coalesce(email, '')) = 'jasonb@neearth.com' then 'admin' else 'viewer' end
from auth.users
on conflict (user_id) do update
set email = excluded.email,
    role = case when lower(excluded.email) = 'jasonb@neearth.com' then 'admin' else public.equipment_material_user_roles.role end,
    updated_at = now();
